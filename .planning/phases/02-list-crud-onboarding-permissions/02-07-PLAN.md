```dart
import 'package:flutter/material.dart';
import 'package:go_router/go_router.dart';
import 'package:not_to_do_list/data/database/app_database.dart';
import 'package:not_to_do_list/features/list/widgets/app_icon.dart';

/// 64 dp compact row. Icon source distinguishes Apps (AppIcon for packageName)
/// from Habits (Icons.spa_outlined). No badges, no tags, no sectioning.
/// CONTEXT.md "App vs Habit visual distinction: Icon source only."
class BlockListRow extends StatelessWidget {
  const BlockListRow({super.key, required this.entry});
  final BlockListData entry;

  bool get _isApp => entry.kind == 0;

  @override
  Widget build(BuildContext context) {
    final cs = Theme.of(context).colorScheme;
    final tt = Theme.of(context).textTheme;
    return SizedBox(
      height: 64,
      child: ListTile(
        minVerticalPadding: 0,
        contentPadding: const EdgeInsets.symmetric(horizontal: 16),
        tileColor: cs.surfaceContainer,
        leading: SizedBox(
          width: 48, height: 48,
          child: Center(
            child: _isApp && entry.packageName != null
                ? AppIcon(packageName: entry.packageName!)
                : Icon(Icons.spa_outlined, size: 32, color: cs.onSurfaceVariant),
          ),
        ),
        title: Text(
          entry.displayName,
          style: tt.titleMedium,
          maxLines: 1,
          overflow: TextOverflow.ellipsis,
        ),
        subtitle: entry.reasonNote.isEmpty
            ? null
            : Text(
                entry.reasonNote,
                style: tt.bodyMedium?.copyWith(color: cs.onSurfaceVariant),
                maxLines: 1,
                overflow: TextOverflow.ellipsis,
              ),
        trailing: Text(
          '—', // Phase 2 streak placeholder; Phase 5 STRK-01/07 fills the integer.
          style: tt.labelSmall?.copyWith(color: cs.onSurfaceVariant),
        ),
        onTap: () => context.go('/list/edit/${entry.id}'),
        // No onLongPress — CONTEXT.md "Long-press has no behavior in v1."
      ),
    );
  }
}
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:go_router/go_router.dart';
import 'package:not_to_do_list/domain/providers/block_list_repo_provider.dart';
import 'package:not_to_do_list/features/home/widgets/block_list_row.dart';
import 'package:not_to_do_list/features/home/widgets/empty_home_state.dart';

class HomeScreen extends ConsumerWidget {
  const HomeScreen({super.key});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final repo = ref.watch(blockListRepoProvider);
    return Scaffold(
      appBar: AppBar(
        title: const Text('Not To-Do List'),
        centerTitle: false,
      ),
      body: StreamBuilder(
        stream: repo.watchAll(),
        builder: (context, snap) {
          if (!snap.hasData) {
            return const SizedBox.shrink();
          }
          final entries = snap.data!;
          // Plan 02-09 will inject the HealthCheckBanner ABOVE this list.
          if (entries.isEmpty) return const EmptyHomeState();
          return ListView.builder(
            itemCount: entries.length,
            itemBuilder: (ctx, i) => BlockListRow(entry: entries[i]),
          );
        },
      ),
      floatingActionButton: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          FloatingActionButton.extended(
            heroTag: 'add_habit',
            onPressed: () => context.go('/list/add-habit'),
            label: const Text('+ Add habit'),
            backgroundColor: Theme.of(context).colorScheme.surface,
            foregroundColor: Theme.of(context).colorScheme.primary,
          ),
          const SizedBox(height: 12),
          FloatingActionButton.extended(
            heroTag: 'add_app',
            onPressed: () => context.go('/list/add-app'),
            label: const Text('+ Add app'),
            backgroundColor: Theme.of(context).colorScheme.primary,
            foregroundColor: Theme.of(context).colorScheme.onPrimary,
          ),
        ],
      ),
      floatingActionButtonLocation: FloatingActionButtonLocation.endFloat,
    );
  }
}
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:go_router/go_router.dart';
import 'package:mocktail/mocktail.dart';
import 'package:not_to_do_list/data/database/app_database.dart';
import 'package:not_to_do_list/data/repositories/block_list_repository.dart';
import 'package:not_to_do_list/domain/providers/block_list_repo_provider.dart';
import 'package:not_to_do_list/features/home/pages/home_screen.dart';

class _FakeRepo extends Mock implements BlockListRepository {}

void main() {
  group('HomeScreen unified list (LIST-06)', () {
    late _FakeRepo repo;

    setUp(() {
      repo = _FakeRepo();
    });

    Widget _harness({required Stream<List<BlockListData>> stream}) {
      when(() => repo.watchAll()).thenAnswer((_) => stream);
      return ProviderScope(
        overrides: [blockListRepoProvider.overrideWith((_) => repo)],
        child: MaterialApp.router(
          routerConfig: GoRouter(routes: [
            GoRoute(path: '/', builder: (_, __) => const HomeScreen()),
            GoRoute(path: '/list/add-app', builder: (_, __) => const Scaffold(body: Text('add-app'))),
            GoRoute(path: '/list/add-habit', builder: (_, __) => const Scaffold(body: Text('add-habit'))),
          ]),
        ),
      );
    }

    BlockListData _entry({required int id, required int kind, String name = 'X', String? pkg, String reason = ''}) {
      final now = DateTime.utc(2026, 5, 5);
      return BlockListData(
        id: id, kind: kind, packageName: pkg, displayName: name,
        reasonNote: reason, streakBreakThresholdMinutes: 5,
        createdAt: now, updatedAt: now, blockMode: 'soft',
        scheduleStartMinutes: null, scheduleEndMinutes: null, scheduleWeekdayMask: null,
      );
    }

    testWidgets('empty list shows EmptyHomeState message', (tester) async {
      await tester.pumpWidget(_harness(stream: Stream.value(const <BlockListData>[])));
      await tester.pumpAndSettle();
      expect(find.text('Nothing on your list yet.'), findsOneWidget);
      expect(find.text('Add an app or habit you want to avoid to get started.'), findsOneWidget);
    });

    testWidgets('renders unified list with both apps and habits in single ListView', (tester) async {
      final entries = [
        _entry(id: 1, kind: 0, name: 'Instagram', pkg: 'com.instagram.android', reason: 'Doomscrolling'),
        _entry(id: 2, kind: 1, name: 'Checking news'),
      ];
      await tester.pumpWidget(_harness(stream: Stream.value(entries)));
      await tester.pumpAndSettle();
      expect(find.text('Instagram'), findsOneWidget);
      expect(find.text('Checking news'), findsOneWidget);
      // No section headers
      expect(find.text('Apps'), findsNothing);
      expect(find.text('Habits'), findsNothing);
    });

    testWidgets('+ Add app FAB and + Add habit FAB both render', (tester) async {
      await tester.pumpWidget(_harness(stream: Stream.value(const <BlockListData>[])));
      await tester.pumpAndSettle();
      expect(find.text('+ Add app'), findsOneWidget);
      expect(find.text('+ Add habit'), findsOneWidget);
    });

    testWidgets('streak placeholder is em-dash for Phase 2', (tester) async {
      final entries = [_entry(id: 1, kind: 0, name: 'X', pkg: 'com.x')];
      await tester.pumpWidget(_harness(stream: Stream.value(entries)));
      await tester.pumpAndSettle();
      expect(find.text('—'), findsOneWidget);
    });
  });
}
```
