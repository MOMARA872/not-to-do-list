```dart
import 'package:drift/drift.dart';
import 'package:drift/native.dart';
import 'package:drift_dev/api/migrations.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:not_to_do_list/data/database/app_database.dart';
import '../../generated_migrations/schema.dart';
import '../../generated_migrations/schema_v1.dart' as v1;

void main() {
  group('AppDatabase v1 → v2 migration (LIST-08, LIST-09)', () {
    late SchemaVerifier verifier;

    setUpAll(() {
      verifier = SchemaVerifier(GeneratedHelper());
    });

    test('schemaVersion is 2', () async {
      final db = AppDatabase(NativeDatabase.memory());
      addTearDown(db.close);
      expect(db.schemaVersion, 2);
    });

    test('v1 → v2: existing v1 row gets block_mode=soft and null schedules', () async {
      // Open at v1 using generated helper, insert a row, then upgrade to v2.
      final connection = await verifier.startAt(1);
      final v1Db = v1.DatabaseAtV1(connection);
      await v1Db.customStatement(
        "INSERT INTO block_list (kind, display_name, reason_note, "
        "streak_break_threshold_minutes, created_at, updated_at) "
        "VALUES (0, 'Instagram', 'Doomscrolling', 5, ?, ?)",
        [DateTime.utc(2026, 4, 27).millisecondsSinceEpoch ~/ 1000,
         DateTime.utc(2026, 4, 27).millisecondsSinceEpoch ~/ 1000],
      );
      await v1Db.close();

      // Reopen via the new schema; Drift triggers onUpgrade.
      final db = AppDatabase(connection);
      addTearDown(db.close);
      final rows = await db.select(db.blockList).get();
      expect(rows, hasLength(1));
      expect(rows[0].displayName, 'Instagram');
      expect(rows[0].blockMode, 'soft');
      expect(rows[0].scheduleStartMinutes, isNull);
      expect(rows[0].scheduleEndMinutes, isNull);
      expect(rows[0].scheduleWeekdayMask, isNull);
    });

    test('v2 createAll round-trip preserves schedule columns', () async {
      final db = AppDatabase(NativeDatabase.memory());
      addTearDown(db.close);
      await db.into(db.blockList).insert(BlockListCompanion.insert(
        kind: 0,
        packageName: const Value('com.tiktok'),
        displayName: 'TikTok',
        reasonNote: const Value(''),
        createdAt: DateTime.utc(2026, 5, 5),
        updatedAt: DateTime.utc(2026, 5, 5),

        scheduleEndMinutes: const Value(360),     // 06:00 (cross-midnight)

      final rows = await db.select(db.blockList).get();
      expect(rows.single.blockMode, 'hard');
      expect(rows.single.scheduleStartMinutes, 1320);
      expect(rows.single.scheduleEndMinutes, 360);
      expect(rows.single.scheduleWeekdayMask, 0x1F);
```

Step 3: Replace the Wave 0 stub in `test/data/repositories/cascade_delete_test.dart` (LIST-05). Use the in-memory `AppDatabase`:

```dart
import 'package:drift/drift.dart';
import 'package:drift/native.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:not_to_do_list/data/database/app_database.dart';

void main() {
  group('BlockList cascade delete (LIST-05)', () {
    late AppDatabase db;
    setUp(() => db = AppDatabase(NativeDatabase.memory()));
    tearDown(() => db.close());

    test('deleting a block_list row removes child pause_events rows', () async {
      // Insert parent
      final parentId = await db.into(db.blockList).insert(BlockListCompanion.insert(
        kind: 0,
        packageName: const Value('com.x'),
        displayName: 'X',
        createdAt: DateTime.utc(2026, 5, 5),
        updatedAt: DateTime.utc(2026, 5, 5),
      ));
      // Insert child pause_events row referencing parent
      await db.customStatement(
        "INSERT INTO pause_events (entry_id, occurred_at, action) VALUES (?, ?, ?)",
        [parentId, DateTime.utc(2026, 5, 5).millisecondsSinceEpoch ~/ 1000, 'cancel'],
      );
      // Sanity: child exists
      final childrenBefore = await db.customSelect(
        "SELECT COUNT(*) AS c FROM pause_events WHERE entry_id = ?",
        variables: [Variable<int>(parentId)],
      ).getSingle();
      expect(childrenBefore.read<int>('c'), 1);

      // Delete parent
      await (db.delete(db.blockList)..where((t) => t.id.equals(parentId))).go();

      // Child must be gone
      final childrenAfter = await db.customSelect(
        "SELECT COUNT(*) AS c FROM pause_events WHERE entry_id = ?",
        variables: [Variable<int>(parentId)],
      ).getSingle();
      expect(childrenAfter.read<int>('c'), 0);
    });
  });
}
```

Important: the exact column names in `pause_events` (entry_id, occurred_at, action) are inherited from Phase 1 — verify them by reading `lib/data/database/tables/pause_events_table.dart` and adjust the raw SQL if Phase 1 used different names.

Both new tests must run in <2 seconds combined (they use NativeDatabase.memory()).
